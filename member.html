function logout() { localStorage.removeItem("loginUser"); window.location.href = "index.html"; } // 当前选中赛季，默认全部比赛 let currentSeason = "all"; const seasonPanel = document.getElementById("seasonPanel"); const seasonSelect = document.getElementById("seasonSelect"); window.onload = function() { // 从本地存储读取真实登录账号 const userName = localStorage.getItem("loginUser"); // 没有登录 → 跳转登录页 if(!userName){ window.location.href = "index.html"; return; } loadPlayerStats(); // 下拉切换监听 seasonSelect.addEventListener("change", function() { currentSeason = this.value; loadPlayerStats(); }) } // 拉取数据并渲染 async function loadPlayerStats() { const userName = localStorage.getItem("loginUser"); // 重置所有显示为-- document.getElementById("kdVal").innerText = "--"; document.getElementById("killVal").innerText = "--"; document.getElementById("matchVal").innerText = "--"; document.getElementById("winVal").innerText = "--"; document.getElementById("dmgVal").innerText = "--"; document.getElementById("top10Val").innerText = "--%"; try { // 携带赛季参数请求接口 const res = await fetch(`/api/playerData?username=${userName}&season=${currentSeason}`); const json = await res.json(); if (!json.ok || !json.data) return; const info = json.data; // 渲染顶部生涯四块数据（汇总/单赛季通用） if (info.historical_total_kd && info.historical_total_kd > 0) { document.getElementById("kdVal").innerText = info.historical_total_kd; } else { document.getElementById("kdVal").innerText = "--"; } document.getElementById("killVal").innerText = info.total_kill ?? 0; document.getElementById("matchVal").innerText = info.total_match ?? 0; document.getElementById("winVal").innerText = info.total_win ?? 0; // 控制本赛季面板显示隐藏 if (currentSeason === "all") { seasonPanel.style.display = "none"; } else { seasonPanel.style.display = "block"; // 单赛季计算场均伤害、前十率 const match = Number(info.total_match) || 0; if (match > 0) { // 场均伤害取整 const avgDmg = Math.round(Number(info.total_damage) / match); document.getElementById("dmgVal").innerText = avgDmg; // 前十率（吃鸡/总场次） const winRate = ((Number(info.total_win) / match) * 100).toFixed(1); document.getElementById("top10Val").innerText = winRate + "%"; } } } catch (err) { console.error("数据加载失败：", err); } }
