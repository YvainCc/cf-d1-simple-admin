const usernameInput = document.getElementById('username'); const pwdInput = document.getElementById('pwd'); const submitBtn = document.getElementById('submitBtn'); const errorMsg = document.getElementById('errorMsg'); submitBtn.addEventListener('click', loginSubmit); document.addEventListener('keydown', (e) => { if(e.key === 'Enter') loginSubmit(); }) async function loginSubmit() { const username = usernameInput.value.trim(); const password = pwdInput.value.trim(); errorMsg.style.display = 'none'; if (!username || !password) { showErr('账号密码不能为空'); return; } try { const res = await fetch('https://dmn-api.yvainchuang.workers.dev/api/login', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ username: username, password: password }) }) const data = await res.json(); if (data.ok) { localStorage.setItem("loginUser", username); // 区分跳转，后端返回没有role，统一先跳队员中心 window.location.href = 'member.html'; } else { showErr(data.msg); } } catch (err) { showErr('网络请求失败，请稍后重试'); console.error(err); } } function showErr(text) { errorMsg.innerText = '服务异常：' + text; errorMsg.style.display = 'block'; }
